  function parseTimeToMs(timeStr) {
      timeStr = timeStr.trim();
      let match = timeStr.match(/(\d{2}):(\d{2}):(\d{2})\.(\d{3})/);
      if (match) return (parseInt(match[1])*3600000) + (parseInt(match[2])*60000) + (parseInt(match[3])*1000) + parseInt(match[4]);
      match = timeStr.match(/(\d{2}):(\d{2}):(\d{2})/);
      if (match) return (parseInt(match[1])*3600000) + (parseInt(match[2])*60000) + (parseInt(match[3])*1000);
      match = timeStr.match(/(\d{2}):(\d{2})/);
      if (match) return (parseInt(match[1])*3600000) + (parseInt(match[2])*60000);
      return Infinity;
  }

  function extractRowsWithTime(html) {
      if (!html) return [];
      const parser = new DOMParser();
      const doc = parser.parseFromString(html, 'text/html');
      const table = doc.querySelector('table');
      if (!table) return [];
      const tbody = table.querySelector('tbody');
      if (!tbody) return [];
      const rows = [];
      for (let row of tbody.children) {
          const firstTd = row.querySelector('td:first-child');
          if (firstTd) {
              const timeText = firstTd.textContent.trim();
              const timeMs = parseTimeToMs(timeText);
              rows.push({ html: row.outerHTML, timeMs: timeMs });
          } else {
              rows.push({ html: row.outerHTML, timeMs: Infinity });
          }
      }
      return rows;
  }

  function renderCombinedTable() {
      if (currentMode !== 'combined') return;
      if (selectedTables.size === 0) {

          return;
      }
      let allRows = [];
      for (let i = 1; i <= 7; i++) {
          if (selectedTables.has(i)) {
              const rows = extractRowsWithTime(tables[i]);
              allRows.push(...rows);
          }
      }
      // Сортировка по времени (от ранних к поздним)
      allRows.sort((a, b) => a.timeMs - b.timeMs);
      const rowsHtml = allRows.map(row => row.html).join('');
      const html = `
          <table class="combined-table">
              <thead><tr><th>Time</th><th>Type</th><th>Info</th></tr></thead>
              <tbody>${rowsHtml}</tbody>
          </table>
          <div style="font-size: 12px; color: gray; margin-top: 8px;">Отсортировано по времени ↑</div>
      `;
      container.innerHTML = html;
  }

    function showTooltip(e, point) {
      if (point) {
        tooltip.textContent = point.info;
        tooltip.classList.remove('hidden');
        let left = e.clientX + 15;
        let top = e.clientY + 15;
        // профилактика выхода за правый/нижний край экрана
        if (left + tooltip.offsetWidth > window.innerWidth) {
          left = e.clientX - tooltip.offsetWidth - 8;
        }
        if (top + tooltip.offsetHeight > window.innerHeight) {
          top = e.clientY - tooltip.offsetHeight - 8;
        }
        tooltip.style.left = left + 'px';
        tooltip.style.top = top + 'px';
      } else {
        tooltip.classList.add('hidden');
      }
    }

    function findHoveredPoint(mx, my) {
      for (let p of canvasPoints) {
        const dx = mx - p.x;
        const dy = my - p.y;
        if (Math.hypot(dx, dy) <= 6) {   // радиус срабатывания — 6px
          return p;
        }
      }
      return null;
    }

    function getMouseCanvasCoords(e) {
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;
      const mouseX = (e.clientX - rect.left) * scaleX;
      const mouseY = (e.clientY - rect.top) * scaleY;
      return { mx: mouseX, my: mouseY };
    }

    function addMinutes(timeStr, minutesToAdd) {
        let [hours, minutes] = timeStr.split(':').map(Number);
        let totalMinutes = hours * 60 + minutes + minutesToAdd;
        
        let days = Math.floor(totalMinutes / (24 * 60));
        let remaining = totalMinutes % (24 * 60);
        
        // Корректировка отрицательных остатков
        if (remaining < 0) {
            days = Math.ceil(totalMinutes / (24 * 60)); // может быть -1, -2...
            remaining = totalMinutes - days * 24 * 60;
            if (remaining < 0) remaining += 24 * 60;
        }
        
        let newHours = Math.floor(remaining / 60);
        let newMinutes = remaining % 60;
        
        let timePart = `${newHours.toString().padStart(2, '0')}:${newMinutes.toString().padStart(2, '0')}`;
        
        if (days === 0) return timePart;
        if (days > 0) return `${days} д. ${timePart}`;
        return `${days} д. ${timePart}`; // для отрицательных дней
    }
